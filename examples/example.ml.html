4 module type LL = sig
5 
6   type elt
7   type llist
8 
9   (*@ function len (l: llist) : int *)
10 
11   val length : llist -> int
12   (*@ r = length l
13       ensures r = len l *)
14 
15   val is_empty : llist -> bool [@@logic]
16   (*@ r = is_empty l 
17       ensures r <-> len l = 0 *)
18 
19   val get : llist -> int -> elt [@@logic]
20   (*@ r = get l n
21       requires 0 <= n < len l *)
22 
23   (*@ function mem (x: elt) (l: llist) : bool *)
24 
25   val mem' : elt -> llist -> bool
26   (*@ r = mem x l
27       ensures r <-> mem x l *)
28 
29   (*@ predicate reversed (l1 l2: llist) *)
30 
31 	(*@ axiom reversed_length :
32 			forall l1 l2: llist. reversed l1 l2 -> len l1 = len l2 *)
33 
34 	(*@ axiom reversed_elems :
35 	    forall l1 l2: llist. reversed l1 l2 ->
36 				forall i. 0 <= i < len l1 -> 
37 					get l1 i = get l2 (len l1 - 1 - i) *)
38 
39   val rev : llist -> llist
40   (*@ r = rev l
41       ensures reversed r l *)
42 
43   (*@ predicate appended (r l1 l2: llist) *)
44 
45 	(*@ axiom appended_length : 
46 			forall r l1 l2: llist. appended r l1 l2 ->
47 				len r = len l1 + len l2 *)
48 
49 	(*@ axiom appended_order : 
50 			forall r l1 l2: llist. appended r l1 l2 ->
51         forall i. 0 <= i < len l1 -> get r i = get l1 i &&
52         forall i. len l1 <= i < len l1 + len l2 ->
53           get r i = get l2 (i - len l1) *)
54

56   (*@ r = append l1 l2
57       ensures appended r l1 l2 *)
58

60 	(*@ r = map f l
61 			ensures len r = len l
62 			ensures forall i. 0 <= i < len l -> (f (get l i)) = (get r i) *)
63 
64   (*@ predicate filtered (f: elt -> bool) (r l: llist) *)
65

92 module type ComparableType = sig
93 	type t
94 
95 	val eq: t -> t -> bool [@@logic]
96 	(*@ b = eq x y
97 		ensures b <-> x = y *)
98 end

100 module LinkedList (E: ComparableType) : LL = struct
101 
102   type elt = E.t
103   type llist = Nil | Cons of elt * llist
104 
105   (*@ function rec len (l: llist) : int =
106       match l with
107       | Nil -> 0
108       | Cons _ t -> 1 + len t *)
109   (*@ variant l 
110       ensures l = Nil <-> result = 0
111       ensures l <> Nil <-> result > 0 *)
112 
113   let length l = 
114     let rec aux c = function
115       | Nil -> c
116       | Cons (_, t) -> aux (c+1) t
117     (*@ r = aux c s
118         variant s
119         ensures c + len s = r *)
120     in aux 0 l
121   (*@ r = len l
122       ensures r = len l *)
123 
124   let[@logic] is_empty = function
125     | Nil -> true
126     | _ -> false
127   (*@ r = is_empty l 
128       ensures r <-> l = Nil *)
129 
130   let[@logic] rec get l n = 
131     match l with
132     | Nil -> assert false
133     | Cons (h, t) -> 
134       if n = 0 then h 
135       else get t (n-1)
136   (*@ r = get l n
137       variant n
138       requires 0 <= n < len l *)
139 
140   (*@ function rec mem (x: elt) (l: llist) : bool =
141       match l with
142       | Nil -> false
143       | Cons h t -> E.eq h x || mem x t *)
144   (*@ variant l *)
145 
146   let[@logic] mem' x l =
147     let rec aux x = function
148       | Nil -> false
149       | Cons (h, t) -> 
150         if E.eq h x then true
151         else aux x t
152     (*@ r = aux x l'
153         variant l'
154         ensures r <-> mem x l' *)
155     in aux x l
156   (*@ r = mem' x l
157       ensures r <-> mem x l *)
158 
159   (*@ predicate reversed_sub (r s: llist) = 
160       len s <= len r &&
161       forall i. 0 <= i < len s -> 
162         (get r i) = (get s (len s - 1 - i)) *)
163 
164   (*@ predicate valid_accumulator (r acc s: llist) =
165       len r = len acc + len s &&
166       forall i. len s <= i < len r -> 
167         (get r i) = (get acc (i - len s)) *)
168     
169   (*@ predicate reversed (l1 l2: llist) = 
170       len l1 = len l2 && 
171       forall i. 0 <= i < len l1 -> 
172         get l1 i = get l2 (len l1 - 1 - i) *)
173 
174   let[@logic] rev l =
175     let rec aux acc = function
176       | Nil -> acc
177       | Cons (h, t) -> aux (Cons (h, acc)) t
178     (*@ r = aux acc s
179         ensures len r = len acc + len s 
180         ensures reversed_sub r s
181         ensures valid_accumulator r acc s
182         variant s *)
183     in aux Nil l
184   (*@ r = rev l
185       ensures reversed r l *)
186 
187   (*@ predicate appended_sub (r s1: llist) =
188       len s1 <= len r &&
189       forall i. 0 <= i < len s1 -> 
190         get r i = get s1 (len s1 - 1 - i) *)
191 
192   (*@ predicate appended_right (r l1 l2: llist) = 
193       forall i. len l1 <= i < len l1 + len l2 ->
194         get r i = get l2 (i - len l1) *)
195 
196   (*@ predicate appended (r l1 l2: llist) =
197       len r = len l1 + len l2 &&
198       forall i. 0 <= i < len l1 -> get r i = get l1 i &&
199       appended_right r l1 l2 *)
200 
201   let[@logic] append l1 l2 =
202     let rec aux s1 s2 =
203       match s1 with
204       | Nil -> s2
205       | Cons (h, t) -> aux t (Cons (h, s2)) 
206     (*@ r = aux s1 s2
207         variant s1
208         ensures len r = len s1 + len s2
209         ensures appended_sub r s1
210         ensures appended_right r s1 s2 *)
211     in let r1 = rev l1 in aux r1 l2
212   (*@ r = append l1 l2
213       ensures appended r l1 l2 *)
214 
215   (*@ predicate mapped_sub (f: elt -> elt) (r s: llist) = 
216       len s <= len r &&
217       forall i. 0 <= i < len s -> 
218         (get r i) = (f (get s (len s - 1 - i))) *)
219 
220 	let map f l =
221 	let rec aux acc f = function
222 		| Nil -> acc
223 			| Cons (h, t) -> aux (Cons (f h, acc)) f t
224 		(*@ r = aux acc f s
225 				variant s
226 				ensures len r = len acc + len s
227         ensures mapped_sub f r s
228         ensures valid_accumulator r acc s *)
229 		in let l' = rev l in aux Nil f l'	
230 	(*@ r = map f l
231 			ensures len r = len l
232 			ensures forall i. 0 <= i < len l -> (f (get l i)) = (get r i) *)
233 
234   (*@ predicate filtered_relative_order (r l: llist) = 
235       forall i j. 0 <= i <= j < len r -> 
236 				exists i' j'. i <= i' <= j' < len l ->
237 				(get r i) = (get l i') && (get r j) = (get l j') *)
238 
239   (*@ predicate filtered_sub (f: elt -> bool) (r acc s: llist) =
240       0 <= len r <= len acc + len s &&
241       forall x: elt. mem x r -> f x &&
242       filtered_relative_order r s *)
243 
244   (*@ predicate filtered (f: elt -> bool) (r l: llist) = 
245       0 <= len r <= len l &&
246       forall x: elt. mem x r -> f x &&
247       filtered_relative_order r l *)
248 
249 	let filter f l =
250 		let rec aux acc f = function
251 			| Nil -> acc
252 			| Cons (h, t) -> 
253 				if f h then aux (Cons (h, acc)) f t
254 				else aux acc f t 
255 		(*@ r = aux acc f s 
256 				variant s
257 				requires forall x: elt. mem x acc -> f x
258 				ensures filtered_sub f r acc s *)	
259 		in let l' = rev l in aux Nil f l'
260 	(*@ r = filter f l
261 			ensures filtered f r l *)
262 
263   let init s f =
264     let rec aux acc n f =
265       if n = 0 then acc
266       else aux (Cons (f (n-1), acc)) (n-1) f
267     (*@ r = aux acc n f
268         variant n
269         requires 0 <= n <= s
270         requires s = len acc + n 
271         requires forall k. 0 <= k < len acc -> (get acc k) = (f (k+n))
272         ensures len r = len acc + n
273         ensures forall k. 0 <= k < len r -> (get r k) = (f k) *)
274     in aux Nil s f
275   (*@ r = init s f
276       requires s >= 0
277       ensures len r = s
278       ensures forall k. 0 <= k < s -> (get r k) = (f k) *)
279 
280 end
