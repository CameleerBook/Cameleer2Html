5module type LL = sig
6
7  type elt
8  type llist
9
10  (*@ function len (l: llist) : int *)
11
12  val length : llist -> int
13  (*@ r = length l
14      ensures r = len l *)
15
16  val is_empty : llist -> bool [@@logic]
17  (*@ r = is_empty l 
18      ensures r <-> len l = 0 *)
19
20  val get : llist -> int -> elt [@@logic]
21  (*@ r = get l n
22      requires 0 <= n < len l *)
23
24  (*@ function mem (x: elt) (l: llist) : bool *)
25
26  val mem' : elt -> llist -> bool
27  (*@ r = mem x l
28      ensures r <-> mem x l *)
29
30  (*@ predicate reversed (l1 l2: llist) *)
31
32	(*@ axiom reversed_length :
33			forall l1 l2: llist. reversed l1 l2 -> len l1 = len l2 *)
34
35	(*@ axiom reversed_elems :
36	    forall l1 l2: llist. reversed l1 l2 ->
37				forall i. 0 <= i < len l1 -> 
38					get l1 i = get l2 (len l1 - 1 - i) *)
39
40  val rev : llist -> llist
41  (*@ r = rev l
42      ensures reversed r l *)
43
44  (*@ predicate appended (r l1 l2: llist) *)
45
46	(*@ axiom appended_length : 
47			forall r l1 l2: llist. appended r l1 l2 ->
48				len r = len l1 + len l2 *)
49
50	(*@ axiom appended_order : 
51			forall r l1 l2: llist. appended r l1 l2 ->
52        forall i. 0 <= i < len l1 -> get r i = get l1 i &&
53        forall i. len l1 <= i < len l1 + len l2 ->
54          get r i = get l2 (i - len l1) *)
55
56  val append : llist -> llist -> llist
57  (*@ r = append l1 l2
58      ensures appended r l1 l2 *)
59
60	val map : (elt -> elt) -> llist -> llist
61	(*@ r = map f l
62			ensures len r = len l
63			ensures forall i. 0 <= i < len l -> (f (get l i)) = (get r i) *)
64
65  (*@ predicate filtered (f: elt -> bool) (r l: llist) *)
66
67	(*@ axiom filtered_length : 
68			forall f: (elt -> bool), r l: llist. filtered f r l ->
69				0 <= len r <= len l *)
70
71	(*@ axiom filtered_valid :
72			forall f: (elt -> bool), r l: llist. filtered f r l ->
73				forall x: elt. mem x r -> f x *)
74
75  (*@ axiom filtered_relative_pos : 
76      forall f: (elt -> bool), r l: llist. filtered f r l ->
77        forall i j. 0 <= i <= j < len r -> 
78				exists i' j'. i <= i' <= j' < len l ->
79				  (get r i) = (get l i') && (get r j) = (get l j') *)

93module type ComparableType = sig
94	type t
95
96	val eq: t -> t -> bool [@@logic]
97	(*@ b = eq x y
98		ensures b <-> x = y *)
99end

101module LinkedList (E: ComparableType) : LL = struct
102
103  type elt = E.t
104  type llist = Nil | Cons of elt * llist
105
106  (*@ function rec len (l: llist) : int =
107      match l with
108      | Nil -> 0
109      | Cons _ t -> 1 + len t *)
110  (*@ variant l 
111      ensures l = Nil <-> result = 0
112      ensures l <> Nil <-> result > 0 *)
113
114  let length l = 
115    let rec aux c = function
116      | Nil -> c
117      | Cons (_, t) -> aux (c+1) t
118    (*@ r = aux c s
119        variant s
120        ensures c + len s = r *)
121    in aux 0 l
122  (*@ r = len l
123      ensures r = len l *)
124
125  let[@logic] is_empty = function
126    | Nil -> true
127    | _ -> false
128  (*@ r = is_empty l 
129      ensures r <-> l = Nil *)
130
131  let[@logic] rec get l n = 
132    match l with
133    | Nil -> assert false
134    | Cons (h, t) -> 
135      if n = 0 then h 
136      else get t (n-1)
137  (*@ r = get l n
138      variant n
139      requires 0 <= n < len l *)
140
141  (*@ function rec mem (x: elt) (l: llist) : bool =
142      match l with
143      | Nil -> false
144      | Cons h t -> E.eq h x || mem x t *)
145  (*@ variant l *)
146
147  let[@logic] mem' x l =
148    let rec aux x = function
149      | Nil -> false
150      | Cons (h, t) -> 
151        if E.eq h x then true
152        else aux x t
153    (*@ r = aux x l'
154        variant l'
155        ensures r <-> mem x l' *)
156    in aux x l
157  (*@ r = mem' x l
158      ensures r <-> mem x l *)
159
160  (*@ predicate reversed_sub (r s: llist) = 
161      len s <= len r &&
162      forall i. 0 <= i < len s -> 
163        (get r i) = (get s (len s - 1 - i)) *)
164
165  (*@ predicate valid_accumulator (r acc s: llist) =
166      len r = len acc + len s &&
167      forall i. len s <= i < len r -> 
168        (get r i) = (get acc (i - len s)) *)
169    
170  (*@ predicate reversed (l1 l2: llist) = 
171      len l1 = len l2 && 
172      forall i. 0 <= i < len l1 -> 
173        get l1 i = get l2 (len l1 - 1 - i) *)
174
175  let[@logic] rev l =
176    let rec aux acc = function
177      | Nil -> acc
178      | Cons (h, t) -> aux (Cons (h, acc)) t
179    (*@ r = aux acc s
180        ensures len r = len acc + len s 
181        ensures reversed_sub r s
182        ensures valid_accumulator r acc s
183        variant s *)
184    in aux Nil l
185  (*@ r = rev l
186      ensures reversed r l *)
187
188  (*@ predicate appended_sub (r s1: llist) =
189      len s1 <= len r &&
190      forall i. 0 <= i < len s1 -> 
191        get r i = get s1 (len s1 - 1 - i) *)
192
193  (*@ predicate appended_right (r l1 l2: llist) = 
194      forall i. len l1 <= i < len l1 + len l2 ->
195        get r i = get l2 (i - len l1) *)
196
197  (*@ predicate appended (r l1 l2: llist) =
198      len r = len l1 + len l2 &&
199      forall i. 0 <= i < len l1 -> get r i = get l1 i &&
200      appended_right r l1 l2 *)
201
202  let[@logic] append l1 l2 =
203    let rec aux s1 s2 =
204      match s1 with
205      | Nil -> s2
206      | Cons (h, t) -> aux t (Cons (h, s2)) 
207    (*@ r = aux s1 s2
208        variant s1
209        ensures len r = len s1 + len s2
210        ensures appended_sub r s1
211        ensures appended_right r s1 s2 *)
212    in let r1 = rev l1 in aux r1 l2
213  (*@ r = append l1 l2
214      ensures appended r l1 l2 *)
215
216  (*@ predicate mapped_sub (f: elt -> elt) (r s: llist) = 
217      len s <= len r &&
218      forall i. 0 <= i < len s -> 
219        (get r i) = (f (get s (len s - 1 - i))) *)
220
221	let map f l =
222	let rec aux acc f = function
223		| Nil -> acc
224			| Cons (h, t) -> aux (Cons (f h, acc)) f t
225		(*@ r = aux acc f s
226				variant s
227				ensures len r = len acc + len s
228        ensures mapped_sub f r s
229        ensures valid_accumulator r acc s *)
230		in let l' = rev l in aux Nil f l'	
231	(*@ r = map f l
232			ensures len r = len l
233			ensures forall i. 0 <= i < len l -> (f (get l i)) = (get r i) *)
234
235  (*@ predicate filtered_relative_order (r l: llist) = 
236      forall i j. 0 <= i <= j < len r -> 
237				exists i' j'. i <= i' <= j' < len l ->
238				(get r i) = (get l i') && (get r j) = (get l j') *)
239
240  (*@ predicate filtered_sub (f: elt -> bool) (r acc s: llist) =
241      0 <= len r <= len acc + len s &&
242      forall x: elt. mem x r -> f x &&
243      filtered_relative_order r s *)
244
245  (*@ predicate filtered (f: elt -> bool) (r l: llist) = 
246      0 <= len r <= len l &&
247      forall x: elt. mem x r -> f x &&
248      filtered_relative_order r l *)
249
250	let filter f l =
251		let rec aux acc f = function
252			| Nil -> acc
253			| Cons (h, t) -> 
254				if f h then aux (Cons (h, acc)) f t
255				else aux acc f t 
256		(*@ r = aux acc f s 
257				variant s
258				requires forall x: elt. mem x acc -> f x
259				ensures filtered_sub f r acc s *)	
260		in let l' = rev l in aux Nil f l'
261	(*@ r = filter f l
262			ensures filtered f r l *)
263
264  let init s f =
265    let rec aux acc n f =
266      if n = 0 then acc
267      else aux (Cons (f (n-1), acc)) (n-1) f
268    (*@ r = aux acc n f
269        variant n
270        requires 0 <= n <= s
271        requires s = len acc + n 
272        requires forall k. 0 <= k < len acc -> (get acc k) = (f (k+n))
273        ensures len r = len acc + n
274        ensures forall k. 0 <= k < len r -> (get r k) = (f k) *)
275    in aux Nil s f
276  (*@ r = init s f
277      requires s >= 0
278      ensures len r = s
279      ensures forall k. 0 <= k < s -> (get r k) = (f k) *)
280
281end
